"""Meaning Match: scored semantic choices and unscored conversational practice."""
import json,time,threading,urllib.request,os,re
from pathlib import Path
from http.server import BaseHTTPRequestHandler,ThreadingHTTPServer
ROOT=Path(__file__).resolve().parent
OUT=ROOT.parents[1]/'outputs'/'word-game';OUT.mkdir(parents=True,exist_ok=True)
EXO=Path(r'/REDACTED_LOCAL_PATH')
STATE={'running':False,'round':0,'results':[],'phase':'ready'}
LESSONS=[]
# Hand-authored answer key; scenarios are fictional exercises, never biography.
CASES=[
 ('intent','I enjoy making furniture in my spare time.',['A personal disclosure','A request for proof','An order to buy furniture'],0,'Acknowledge personal disclosures without demanding outside evidence.'),
 ('intent','Could you put that in everyday words?',['Speak louder','Explain more simply','Change the subject'],1,'Paraphrase meaning in familiar words when asked to simplify.'),
 ('negation','I like tea, but not with sugar.',['Likes sweet tea','Dislikes all tea','Likes tea without sugar'],2,'Negation changes the detail it applies to, not the whole sentence.'),
 ('reference','Nora gave Eli a book. He thanked her. Who thanked Nora?',['Eli','Nora','The book'],0,'Resolve pronouns using the local sentence context.'),
 ('uncertainty','What color was my first bicycle? No memory is supplied.',['Invent a plausible color','Say you do not know and ask','Refuse all conversation'],1,'Missing memory calls for a relevant question, not invention or refusal.'),
 ('empathy','That was a rough day.',['Demand a citation','Claim you had the same experience','Acknowledge it and invite them to talk'],2,'Empathy does not require claiming matching personal experiences.'),

 ('paraphrase','I am running out of steam.',['The person has a broken boiler','The person feels tired','The person is racing'],1,'Interpret common figurative language using conversational context.'),

def save():
 tmp=OUT/'status.tmp';tmp.write_text(json.dumps(STATE,indent=2),encoding='utf-8');os.replace(tmp,OUT/'status.json')
def request(url,data=None,token=None,timeout=65):
 req=urllib.request.Request(url,data=json.dumps(data).encode() if data else None,headers={'Content-Type':'application/json',**({'Authorization':'Bearer '+token} if token else {})})
 with urllib.request.urlopen(req,timeout=timeout) as r:return json.load(r)
def ask(case,lessons):
 topic,text,choices,answer,lesson=case
 # Rotate options to avoid learning answer positions.
 offset=STATE['round']%3;order=list(range(3));order=order[offset:]+order[:offset]
 displayed=[choices[i] for i in order]
 prompt='You are JARVIS playing Meaning Match. All scenarios are fictional exercises; do not store them as facts about Operator. Choose the best meaning or conversational response, then write one natural short reply. Reply as JSON: {"choice":0 or 1 or 2,"reply":"..."}. No actions or tool calls.\nPractice feedback:\n'+'\n'.join(lessons)+'\nScenario: '+text+'\nChoices: '+json.dumps(displayed)
 config=(EXO/'config.yaml').read_text(encoding='utf-8');model=re.search(r'ollama_model:\s*"([^"]+)"',config).group(1)
 result=request('http://127.0.0.1:11434/api/generate',{'model':model,'prompt':prompt,'stream':False,'think':False,'format':'json','options':{'num_predict':130,'num_ctx':2048}})
 parsed=json.loads(result['response']);choice=parsed.get('choice')
 correct=type(choice)==int and 0<=choice<3 and order[choice]==answer
 return {'topic':topic,'scenario':text,'choices':displayed,'choice':choice,'correct':correct,'expected':choices[answer],'reply':str(parsed.get('reply',''))[:600],'lesson':lesson,'model':model}
def remember():
 try:
  # Reuse the existing package-only game connection; no owner credentials.
  token=(ROOT.parent/'crafter'/'memory-token').read_text().strip()
  summary='Meaning Match semantic practice completed. Fictional scenarios are NOT owner biography. Human-authored communication lessons: '+' '.join(dict.fromkeys(c[5] for c in CASES))
  receipt=request('http://127.0.0.1:18790/packages',{'package_id':'meaning-match-'+str(STATE['started']),'from':'jarvis-crafter','date':time.strftime('%Y-%m-%d'),'requested_action':'note','claims':[{'content':summary,'provenance':'verified','evidence':str(OUT/'results.json')}]},token,25)
  (OUT/'memory-receipt.json').write_text(json.dumps(receipt,indent=2))
  STATE['memory_saved']=bool(receipt.get('complete') and all(x.get('status') in ('APPROVED','NOTED') for x in receipt.get('items',[])))
 except Exception as e:STATE['memory_saved']=False;STATE['memory_error']=str(e)[:200]
def run():
 if STATE['running']:return
 STATE.update(running=True,round=0,results=[],phase='baseline',started=int(time.time()),error=None);LESSONS.clear();save()
 try:
  # Baseline on training items, feedback practice, then different held-out wording.
  for phase,items in [('baseline',CASES[:8]),('practice',CASES[:8]),('transfer',CASES[8:])]:
   STATE['phase']=phase
   for case in items:
    if (ROOT/'stop').exists():STATE['phase']='stopped';return
    deadline=time.time()+600
    while time.time()<deadline:
     health=request('http://127.0.0.1:18790/health',timeout=5)
     voice=EXO/'voice.activity'
     if not health.get('governor_active') and not (voice.exists() and time.time()-voice.stat().st_mtime<60):break
     STATE['waiting']='Waiting for conversation/voice to be idle';save();time.sleep(5)
    else:raise RuntimeError('Jarvis remained busy; practice paused without modifying conversation')
    STATE.pop('waiting',None);STATE['round']+=1
    result=ask(case,LESSONS if phase!='baseline' else []);result['phase']=phase
    STATE['results'].append(result)
    if phase=='practice' and case[5] not in LESSONS:LESSONS.append(case[5])
    save();time.sleep(2)
  STATE['phase']='complete';(OUT/'results.json').write_text(json.dumps(STATE['results'],indent=2),encoding='utf-8');remember()
  scores={p:sum(r['correct'] for r in STATE['results'] if r['phase']==p) for p in ('baseline','practice','transfer')}
  STATE['scores']=scores
  (OUT/'report.md').write_text('# Meaning Match results\n\n'+ '\n'.join(f'- {p}: {s}/8 semantic choices correct.' for p,s in scores.items())+'\n\nReplies are recorded for human review, not automatically graded for naturalness. Transfer uses different wording, but this small hand-authored exercise is not proof of general conversational improvement. No model weights were trained. Only general communication lessons were submitted to shared memory; fictional biographies were not.\n',encoding='utf-8')
 except Exception as e:STATE['error']=str(e)[:400];STATE['phase']='needs_attention'
 finally:STATE['running']=False;save()

class Handler(BaseHTTPRequestHandler):
 def log_message(self,*a):pass
 def do_GET(self):
  if self.path=='/':data=(ROOT/'index.html').read_bytes();kind='text/html'
  elif self.path=='/state':data=json.dumps(STATE).encode();kind='application/json'
  else:self.send_error(404);return
  self.send_response(200);self.send_header('Content-Type',kind);self.send_header('Content-Length',str(len(data)));self.send_header('Cache-Control','no-store');self.end_headers();self.wfile.write(data)
 def do_POST(self):
  if self.headers.get('Origin')!='http://127.0.0.1:18925':self.send_error(403);return
  if self.path=='/stop':(ROOT/'stop').touch()
  else:self.send_error(404);return
  self.send_response(204);self.end_headers()
if __name__=='__main__':
 server=ThreadingHTTPServer(('127.0.0.1',18925),Handler)
 threading.Thread(target=run,daemon=True).start();server.serve_forever()
